import kotlin.random.Random

class SudokuGenerator {
    private val size = 9
    private val subGridSize = 3

    // 難度設定：定義需要挖掉多少個數字（格子總數為 81）
    enum class Difficulty(val holes: Int) {
        EASY(30),      // 剩 51 個提示數
        MEDIUM(42),    // 剩 39 個提示數
        HARD(50),      // 剩 31 個提示數
        EXPERT(54)     // 剩 27 個提示數
    }

    /**
     * 主進入點：生成一個新的數獨謎題與答案
     * @return Pair(謎題矩陣, 答案矩陣)。謎題矩陣中 0 代表空格。
     */
    fun generatePuzzle(difficulty: Difficulty): Pair<Array<IntArray>, Array<IntArray>> {
        val board = Array(size) { IntArray(size) { 0 } }
        
        // 1. 生成一個完整的合法數獨終盤
        fillBoard(board)
        
        // 2. 複製一份作為答案庫
        val solution = Array(size) { board[it].clone() }
        
        // 3. 根據難度進行挖洞
        createHoles(board, difficulty.holes)
        
        return Pair(board, solution)
    }

    // 檢查在 (row, col) 填入 num 是否合法
    private fun isValid(board: Array<IntArray>, row: Int, col: Int, num: Int): Boolean {
        // 檢查行與列
        for (i in 0 until size) {
            if (board[row][i] == num || board[i][col] == num) return false
        }
        
        // 檢查 3x3 九宮格（使用無條件捨去幾何定位公式）
        val startRow = (row / subGridSize) * subGridSize
        val startCol = (col / subGridSize) * subGridSize
        for (i in 0 until subGridSize) {
            for (j in 0 until subGridSize) {
                if (board[startRow + i][startCol + j] == num) return false
            }
        }
        return true
    }

    // 使用回溯法 (Backtracking) 填滿盤面
    private fun fillBoard(board: Array<IntArray>): Boolean {
        for (row in 0 until size) {
            for (col in 0 until size) {
                if (board[row][col] == 0) {
                    // 隨機打亂 1-9 的數字順序，確保每次生成的題目都不同
                    val numbers = (1..9).shuffled()
                    for (num in numbers) {
                        if (isValid(board, row, col, num)) {
                            board[row][col] = num
                            if (fillBoard(board)) return true
                            board[row][col] = 0 // 回溯
                        }
                    }
                    return false // 1-9 都填不進去，觸發上層回溯
                }
            }
        }
        return true
    }

    // 執行挖洞邏輯
    private fun createHoles(board: Array<IntArray>, holesCount: Int) {
        var count = holesCount
        val random = Random.Default
        
        while (count > 0) {
            val cellIndex = random.nextInt(size * size) // 0 到 80
            val row = cellIndex / size
            val col = cellIndex % size
            
            if (board[row][col] != 0) {
                board[row][col] = 0 // 挖空
                count--
            }
        }
        // 註：高階開發通常會在挖洞後用求解器驗證是否為「唯一解」。
        // 此基礎版本已能滿足初階/中階遊戲體驗。
    }
}
